3. GLOBAL CODING GUIDELINES & STYLE RULES

   - JavaScript/Node Syntax: Use modern ES6+ syntax with async/await for async operations.

   - Variable Conventions: Use camelCase for JavaScript variables/functions and snake_case for database fields and SQL queries.

   - Error Handling: Standardized JSON error response format:

     Success: { "success": true, "data": { ... } }

     Failure: { "success": false, "error": "Human-readable error message" }

   - Environment Variables: Never hardcode database credentials or ports. Always reference process.env.



4. DATABASE SCHEMA (POSTGRESQL - SINGLE SOURCE OF TRUTH)



   -- Users Table

   CREATE TABLE users (

       id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

       username VARCHAR(50) NOT NULL,

       friend_code VARCHAR(12) UNIQUE NOT NULL, -- Format: USER-XXXX

       preferred_language VARCHAR(10) DEFAULT 'en',

       created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP

   );



   -- Friendships Table

   CREATE TABLE friendships (

       id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

       requester_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,

       addressee_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,

       status VARCHAR(20) NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'accepted', 'rejected')),

       created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

       updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

       CONSTRAINT unique_friendship UNIQUE (requester_id, addressee_id)

   );



   -- Messages Table

   CREATE TABLE messages (

       id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

       conversation_id VARCHAR(100) NOT NULL,

       sender_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,

       content TEXT NOT NULL,

       source_language VARCHAR(10) NOT NULL,

       is_scheduled BOOLEAN DEFAULT FALSE,

       expires_at TIMESTAMP WITH TIME ZONE NULL,

       created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP

   );



5. SHARED JSON COMMUNICATION CONTRACT

   All API endpoints and WebSocket event payloads MUST strictly follow these exact keys:



   A. User Registration API (/api/users/register)

      Request:  { "username": "JohnDoe", "preferred_language": "en" }

      Response: { "success": true, "user": { "id": "uuid", "username": "JohnDoe", "friend_code": "USER-7F2A", "preferred_language": "en" } }



   B. Friend Request API (/api/friends/request)

      Request:  { "requester_id": "uuid", "target_friend_code": "USER-8A39" }

      Response: { "success": true, "friendship": { "id": "uuid", "status": "pending" } }



   C. WebSocket Message Payload (Event: "send_message" & "receive_message")

      {

        "messageId": "msg_1710000000000",

        "conversationId": "room_userA_userB",

        "senderId": "uuid-of-sender",

        "text": "Hello world",

        "sourceLanguage": "en",

        "delayMs": 0,          // Set to > 0 for scheduled delivery

        "ttlSeconds": 300      // Set to > 0 for auto-deletion countdown

      }



   D. WebSocket Auto-Deletion Event (Event: "message_deleted")

      {

        "conversationId": "room_userA_userB",

        "messageId": "msg_1710000000000"

      }



   E. WebSocket Document Co-Editing Payload (Event: "doc_update")

      {

        "conversationId": "room_userA_userB",

        "docContent": "Updated text shared between users",

        "updatedBy": "uuid-of-sender"

      }
